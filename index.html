async function generateSignal() { const pair = document.getElementById("pair").value; document.getElementById("status").textContent = "LOADING"; document.getElementById("signal").textContent = "WAIT"; try { const response = await fetch( "/.netlify/functions/signal?symbol=" + encodeURIComponent(pair) ); const data = await response.json(); console.log("OTC DATA:", data); document.getElementById("status").textContent = data.status || "ERROR"; document.getElementById("price").textContent = data.price ?? "--"; document.getElementById("strength").textContent = data.strength ? data.strength + "%" : "--"; document.getElementById("signal").textContent = data.signal || "WAIT"; } catch (error) { console.error(error); document.getElementById("status").textContent = "ERROR"; document.getElementById("signal").textContent = "WAIT"; document.getElementById("price").textContent = "--"; document.getElementById("strength").textContent = "--"; } }
